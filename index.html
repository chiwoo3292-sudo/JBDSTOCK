var code = (p.code || '').trim(); var email = (p.email || '').trim(); if (!clientId || !clientSecret || !code || !email) { return jsonResponse({ status: 'error', message: '필수 파라미터 누락(clientId, clientSecret, code, email)' }); } try { var res = UrlFetchApp.fetch('https://oauth2.googleapis.com/token', { method: 'post', muteHttpExceptions: true, payload: { code: code, client_id: clientId, client_secret: clientSecret, redirect_uri: GMAIL_REDIRECT, grant_type: 'authorization_code' } }); var data = JSON.parse(res.getContentText()); if (!data.refresh_token) { return jsonResponse({ status: 'error', message: '토큰 교환 실패: ' + (data.error_description || data.error || res.getContentText()) }); } var props = PropertiesService.getScriptProperties(); props.setProperty('gmail_token_' + email, data.refresh_token); // 연동된 계정 목록에 email 추가 var listRaw = props.getProperty('gmail_accounts') || ''; var list = listRaw ? listRaw.split(',') : []; if (list.indexOf(email) === -1) { list.push(email); props.setProperty('gmail_accounts', list.join(',')); } return jsonResponse({ status: 'ok', email: email }); } catch (err) { return jsonResponse({ status: 'error', message: err.message }); } } // 연동 계정 목록 조회 (토큰 값은 노출하지 않고 email 목록만) function getGmailAccounts(p) { var props = PropertiesService.getScriptProperties(); var listRaw = props.getProperty('gmail_accounts') || ''; var list = listRaw ? listRaw.split(',') : []; return jsonResponse({ status: 'ok', accounts: list }); } // 계정 연동 해제 (토큰 삭제 + 목록에서 제거) function revokeGmailAccount(p) { var email = (p.email || '').trim(); if (!email) return jsonResponse({ status: 'error', message: 'email 없음' }); var props = PropertiesService.getScriptProperties(); props.deleteProperty('gmail_token_' + email); var listRaw = props.getProperty('gmail_accounts') || ''; var list = listRaw ? listRaw.split(',').filter(function(x){ return x !== email; }) : []; props.setProperty('gmail_accounts', list.join(',')); return jsonResponse({ status: 'ok' }); } // ════════════════════════════════════════════════════════════ // jsonResponse - 공통 응답 함수 // ════════════════════════════════════════════════════════════ function jsonResponse(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON);
